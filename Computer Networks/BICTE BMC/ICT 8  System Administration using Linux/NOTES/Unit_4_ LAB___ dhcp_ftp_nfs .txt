=====================================================================
 DHCP SERVER LAB  -  5 Virtual PCs + DORA Verification
 Platform : Killercoda (Ubuntu) or any Ubuntu Linux terminal (run as root)
 Method   : Linux network namespaces (each namespace = 1 virtual PC)
=====================================================================

TOPOLOGY
--------
                 +--------------------------------------+
                 |  Ubuntu host  (DHCP server, dnsmasq) |
                 |  bridge br-dhcp = 192.168.50.1/24    |
                 +----+------+------+------+------+-----+
                      |      |      |      |      |
                     pc1    pc2    pc3    pc4    pc5
                  (each is a namespace with eth0, gets IP via DHCP)

 DHCP pool  : 192.168.50.100 - 192.168.50.150
 Gateway    : 192.168.50.1
 DNS        : 8.8.8.8
 Lease time : 12h


=====================================================================
 STEP 0 : BECOME ROOT
=====================================================================
sudo -i


=====================================================================
 STEP 1 : INSTALL PACKAGES
=====================================================================
apt update
apt install -y dnsmasq isc-dhcp-client tcpdump iproute2

# dnsmasq auto-starts after install; stop it, we will run our own copy
systemctl stop dnsmasq 2>/dev/null
systemctl disable dnsmasq 2>/dev/null








=====================================================================
 STEP 2 : CREATE BRIDGE + 5 VIRTUAL PCs  (copy-paste whole block)
=====================================================================

# --- bridge (acts as the virtual switch + DHCP server interface) ---
ip link add br-dhcp type bridge
ip addr add 192.168.50.1/24 dev br-dhcp
ip link set br-dhcp up



# --- create pc1..pc5 ---
for i in 1 2 3 4 5; do
  ip netns add pc$i
  ip link add veth-pc$i type veth peer name vb-pc$i
  ip link set veth-pc$i netns pc$i
  ip link set vb-pc$i master br-dhcp
  ip link set vb-pc$i up
  ip netns exec pc$i ip link set veth-pc$i name eth0
  ip netns exec pc$i ip link set lo up
  ip netns exec pc$i ip link set eth0 address 02:00:00:00:00:0$i
  ip netns exec pc$i ip link set eth0 up
done

# --- verify the 5 PCs exist ---
ip netns list
ip netns exec pc1 ip -br link









=====================================================================
 STEP 3 : START THE DHCP SERVER
=====================================================================
dnsmasq --conf-file=/dev/null \
  --port=0 \
  --interface=br-dhcp --bind-interfaces \
  --dhcp-range=192.168.50.100,192.168.50.150,255.255.255.0,12h \
  --dhcp-option=option:router,192.168.50.1 \
  --dhcp-option=option:dns-server,8.8.8.8 \
  --dhcp-leasefile=/tmp/dnsmasq.leases \
  --log-dhcp --log-facility=/tmp/dnsmasq.log \
  --pid-file=/tmp/dnsmasq.pid



# check server is running and listening on UDP 67
ps aux | grep [d]nsmasq
ss -ulnp | grep :67

# (port=0 disables DNS so it will not clash with systemd-resolved)





=====================================================================
 STEP 4 : START PACKET CAPTURE FOR DORA  (background)
=====================================================================
# Captures Discover / Offer / Request / Ack on the bridge
tcpdump -i br-dhcp -n -l -e -vv 'udp port 67 or udp port 68' > /tmp/dora_capture.txt 2>&1 &
echo $! > /tmp/tcpdump.pid






=====================================================================
 STEP 5 : GET IP ON ALL 5 PCs  (this triggers DORA)
=====================================================================
for i in 1 2 3 4 5; do
  echo "================ pc$i : DORA ================"
  ip netns exec pc$i dhclient -v \
      -lf /var/lib/dhcp/pc$i.leases \
      -pf /run/dhclient-pc$i.pid eth0
done

# You will see for every PC:
#   DHCPDISCOVER on eth0 ...      (D - client broadcasts)
#   DHCPOFFER of 192.168.50.x     (O - server offers)
#   DHCPREQUEST for 192.168.50.x  (R - client requests)
#   DHCPACK of 192.168.50.x       (A - server acknowledges)
#   bound to 192.168.50.x





=====================================================================
 STEP 6 : VERIFY IPs / GATEWAY / DNS
=====================================================================
# IP of every PC
for i in 1 2 3 4 5; do
  echo "--- pc$i ---"
  ip netns exec pc$i ip -4 -br addr show eth0
done

# default route given by DHCP
ip netns exec pc1 ip route

# server side lease table
cat /tmp/dnsmasq.leases
# format: expiry  MAC  IP  hostname  client-id



# ping test: pc1 -> server, pc1 -> pc2
ip netns exec pc1 ping -c 2 192.168.50.1

PC2_IP=$(ip netns exec pc2 ip -4 -o addr show eth0 | awk '{print $4}' | cut -d/ -f1)


ip netns exec pc1 ping -c 2 $PC2_IP






=====================================================================
 STEP 7 : PROVE DORA  (server log + packet capture)
=====================================================================
# Server log - shows all 4 DORA messages per client
grep -E "DHCPDISCOVER|DHCPOFFER|DHCPREQUEST|DHCPACK" /tmp/dnsmasq.log

# Packet capture - stop it and read
kill $(cat /tmp/tcpdump.pid)
sleep 1
grep -E "DHCP-Message|Discover|Offer|Request|ACK" /tmp/dora_capture.txt | head -40

# Live DORA for one PC (open a 2nd terminal tab and run this FIRST):
#   tcpdump -i br-dhcp -n -vv 'udp port 67 or udp port 68'
# then in 1st tab run the release/renew commands of STEP 8.


=====================================================================
 STEP 8 : RELEASE IP  and  RENEW (DORA AGAIN)
=====================================================================
# ---- RELEASE pc1 (sends DHCPRELEASE) ----
ip netns exec pc1 dhclient -v -r \
    -lf /var/lib/dhcp/pc1.leases \
    -pf /run/dhclient-pc1.pid eth0

ip netns exec pc1 ip -4 addr show eth0      # IP should be gone
cat /tmp/dnsmasq.leases                      # pc1 lease removed
grep DHCPRELEASE /tmp/dnsmasq.log

# ---- RENEW / GET IP AGAIN (full DORA again) ----
ip netns exec pc1 dhclient -v \
    -lf /var/lib/dhcp/pc1.leases \
    -pf /run/dhclient-pc1.pid eth0

ip netns exec pc1 ip -4 -br addr show eth0

# ---- RELEASE + RENEW ALL 5 PCs ----
for i in 1 2 3 4 5; do
  ip netns exec pc$i dhclient -r -lf /var/lib/dhcp/pc$i.leases -pf /run/dhclient-pc$i.pid eth0
done
sleep 2
for i in 1 2 3 4 5; do
  ip netns exec pc$i dhclient -v -lf /var/lib/dhcp/pc$i.leases -pf /run/dhclient-pc$i.pid eth0
done


=====================================================================
 STEP 9 : "UP / DOWN" TEST  (interface down then up = new DORA)
=====================================================================
ip netns exec pc3 ip link set eth0 down
ip netns exec pc3 ip link set eth0 up
ip netns exec pc3 dhclient -v -lf /var/lib/dhcp/pc3.leases -pf /run/dhclient-pc3.pid eth0
ip netns exec pc3 ip -4 -br addr show eth0


=====================================================================
 STEP 10 : ONE-SHOT VERIFY SCRIPT  (save as verify_dhcp.sh)
=====================================================================
cat > /root/verify_dhcp.sh << 'EOF'
#!/bin/bash
echo "===== DHCP SERVER ====="
pgrep -a dnsmasq || echo "dnsmasq NOT running"
ss -ulnp | grep :67 || echo "port 67 NOT listening"
echo
echo "===== CLIENT STATUS ====="
PASS=0
for i in 1 2 3 4 5; do
  IP=$(ip netns exec pc$i ip -4 -o addr show eth0 | awk '{print $4}')
  if [ -n "$IP" ]; then
    echo "pc$i : $IP  [OK]"; PASS=$((PASS+1))
  else
    echo "pc$i : no IP      [FAIL]"
  fi
done
echo
echo "===== LEASES ====="
cat /tmp/dnsmasq.leases
echo
echo "===== DORA COUNT IN SERVER LOG ====="
for m in DHCPDISCOVER DHCPOFFER DHCPREQUEST DHCPACK DHCPRELEASE; do
  printf "%-13s : %s\n" $m "$(grep -c $m /tmp/dnsmasq.log)"
done
echo
echo "RESULT: $PASS/5 clients got an IP"
EOF
chmod +x /root/verify_dhcp.sh
/root/verify_dhcp.sh


=====================================================================
 STEP 11 : CLEANUP  (remove everything)
=====================================================================
for i in 1 2 3 4 5; do
  ip netns exec pc$i dhclient -r -lf /var/lib/dhcp/pc$i.leases -pf /run/dhclient-pc$i.pid eth0 2>/dev/null
  ip netns del pc$i
  rm -f /var/lib/dhcp/pc$i.leases
done
kill $(cat /tmp/dnsmasq.pid) 2>/dev/null
kill $(cat /tmp/tcpdump.pid) 2>/dev/null
ip link del br-dhcp
rm -f /tmp/dnsmasq.* /tmp/dora_capture.txt


=====================================================================
 DORA CHEAT SHEET
=====================================================================
 D - DISCOVER : client -> broadcast (0.0.0.0:68 -> 255.255.255.255:67)
                "Is any DHCP server out there?"
 O - OFFER    : server -> client   "Here is an IP: 192.168.50.1xx"
 R - REQUEST  : client -> broadcast "I accept that IP"
 A - ACK      : server -> client   "Confirmed, lease = 12h"

 Extra messages : DHCPRELEASE (client gives IP back), DHCPNAK,
                  DHCPDECLINE, DHCPINFORM

=====================================================================
 TROUBLESHOOTING
=====================================================================
 * "RTNETLINK: Operation not permitted"  -> you are not root (use sudo -i)
 * dnsmasq: "Address already in use"     -> pkill dnsmasq, run STEP 3 again
 * Client gets no IP                     -> check:  ip link show br-dhcp
                                                    ss -ulnp | grep :67
                                                    tail /tmp/dnsmasq.log
 * dhclient permission/apparmor error    -> keep lease files in /var/lib/dhcp/
                                            exactly as written above
 * Namespace already exists              -> run STEP 11 cleanup first

FTP SERVER (vsftpd) - KILLERCODA UBUNTU - SIMPLE STEPS
=======================================================

• STEP 1 : Become root
    sudo -i

• STEP 2 : Install vsftpd and FTP client tools
    apt update
    apt install -y vsftpd ftp curl

• STEP 3 : Backup the original config
    cp /etc/vsftpd.conf /etc/vsftpd.conf.bak

• STEP 4 : Write the new config
    cat > /etc/vsftpd.conf << 'EOF'
    listen=YES
    listen_ipv6=NO
    anonymous_enable=NO
    local_enable=YES
    write_enable=YES
    local_umask=022
    dirmessage_enable=YES
    use_localtime=YES
    xferlog_enable=YES
    connect_from_port_20=YES
    chroot_local_user=YES
    allow_writeable_chroot=YES
    secure_chroot_dir=/var/run/vsftpd/empty
    pam_service_name=vsftpd
    pasv_enable=YES
    pasv_min_port=40000
    pasv_max_port=40100
    EOF

  (If the lines above keep their leading spaces, use this instead:)
    sed -i 's/^[ \t]*//' /etc/vsftpd.conf

• STEP 5 : Create the FTP user and a test file
    useradd -m -s /bin/bash ftpuser
    echo "ftpuser:ftp123" | chpasswd
    mkdir -p /home/ftpuser/files
    echo "Hello from FTP server" > /home/ftpuser/files/welcome.txt
    chown -R ftpuser:ftpuser /home/ftpuser

• STEP 6 : Start and enable the service
    mkdir -p /var/run/vsftpd/empty
    systemctl restart vsftpd
    systemctl enable vsftpd
    (if systemctl does not work:  service vsftpd restart)





=======================================================
VERIFY CONFIGURATION
=======================================================

• CHECK 1 : Service is running
    systemctl status vsftpd --no-pager
    -> must show:  active (running)

• CHECK 2 : Port 21 is listening
    ss -tlnp | grep :21
    -> must show vsftpd on 0.0.0.0:21

• CHECK 3 : Config values are applied
    grep -v '^#' /etc/vsftpd.conf | grep -E "anonymous_enable|local_enable|write_enable|chroot_local_user|pasv"
    -> anonymous_enable=NO, local_enable=YES, write_enable=YES

• CHECK 4 : Login test (list files)
    curl -u ftpuser:ftp123 ftp://localhost/files/
    -> must list welcome.txt




• CHECK 5 : Download test
    curl -u ftpuser:ftp123 ftp://localhost/files/welcome.txt -o /tmp/downloaded.txt


    cat /tmp/downloaded.txt
    -> must print:  Hello from FTP server



• CHECK 6 : Upload test
    echo "uploaded by client" > /tmp/upload_test.txt

    curl -T /tmp/upload_test.txt -u ftpuser:ftp123 ftp://localhost/files/

    ls -l /home/ftpuser/files/
    -> upload_test.txt must appear



• CHECK 7 : Wrong password must FAIL
    curl -u ftpuser:wrongpass ftp://localhost/
    -> must show:  Login denied (530)


• CHECK 8 : Anonymous login must FAIL
    curl -u anonymous:test ftp://localhost/
    -> must show:  Login denied (530)



• CHECK 9 : Interactive FTP session (optional)
    ftp localhost
      Name: ftpuser
      Password: ftp123
      ftp> ls
      ftp> cd files
      ftp> get welcome.txt
      ftp> put /tmp/upload_test.txt
      ftp> bye

• CHECK 10 : Server log
    tail -n 20 /var/log/vsftpd.log
    -> shows OK LOGIN, OK UPLOAD, OK DOWNLOAD









=======================================================
ONE-SHOT VERIFY SCRIPT
=======================================================
    cat > /root/verify_ftp.sh << 'EOF'
    #!/bin/bash
    U=ftpuser; P=ftp123
    ok(){ echo "[PASS] $1"; }
    no(){ echo "[FAIL] $1"; }
    systemctl is-active --quiet vsftpd && ok "vsftpd running" || no "vsftpd running"
    ss -tln | grep -q ':21 ' && ok "port 21 listening" || no "port 21 listening"
    grep -q '^anonymous_enable=NO' /etc/vsftpd.conf && ok "anonymous disabled" || no "anonymous disabled"
    grep -q '^write_enable=YES' /etc/vsftpd.conf && ok "write enabled" || no "write enabled"
    curl -s -u $U:$P ftp://localhost/files/ | grep -q welcome.txt && ok "login + list" || no "login + list"
    curl -s -u $U:$P ftp://localhost/files/welcome.txt | grep -q "Hello from FTP" && ok "download" || no "download"
    echo test > /tmp/v.txt
    curl -s -T /tmp/v.txt -u $U:$P ftp://localhost/files/ && [ -f /home/$U/files/v.txt ] && ok "upload" || no "upload"
    curl -s -u $U:wrong ftp://localhost/ >/dev/null 2>&1 && no "bad password blocked" || ok "bad password blocked"
    curl -s -u anonymous:x ftp://localhost/ >/dev/null 2>&1 && no "anonymous blocked" || ok "anonymous blocked"
    EOF
    sed -i 's/^[ \t]*//' /root/verify_ftp.sh
    chmod +x /root/verify_ftp.sh
    /root/verify_ftp.sh


=======================================================
TROUBLESHOOTING
=======================================================
• 530 Login incorrect       -> check user/password; check  cat /etc/ftpusers  (ftpuser must not be listed)
• Connection refused        -> service not running:  journalctl -u vsftpd -n 20
• 500 OOPS: config error    -> fix typo in /etc/vsftpd.conf, restart
• Killercoda external access -> use the Traffic/Port tab for port 21 only if offered;
                                 testing on localhost is enough for this lab.

CLEANUP
    systemctl stop vsftpd
    userdel -r ftpuser
    cp /etc/vsftpd.conf.bak /etc/vsftpd.conf

NFS SERVER + WEB FILE VIEWER - KILLERCODA UBUNTU - SIMPLE STEPS
================================================================
Server share : /srv/nfs/share        (exported over NFS)
Client mount : /mnt/nfs-client       (mounted from localhost)
Web viewer   : nginx on port 8080    (shows the files from the NFS mount)



Flow:  upload file -> NFS mount -> stored in server share -> visible in browser




• STEP 1 : Become root and install packages
    sudo -i
    apt update
    apt install -y nfs-kernel-server nfs-common nginx

• STEP 2 : Create the share folder
    mkdir -p /srv/nfs/share
    chmod 777 /srv/nfs/share
    echo "Hello from NFS server" > /srv/nfs/share/welcome.txt

• STEP 3 : Configure the export
    echo "/srv/nfs/share *(rw,sync,no_subtree_check,no_root_squash)" >> /etc/exports
    exportfs -ra


  Meaning of options:
    rw               read and write allowed
    sync             write to disk before replying
    no_subtree_check avoids subtree check problems
    no_root_squash   root on client keeps root rights (lab only)
    *                any client (in real life use a network, e.g. 192.168.1.0/24)



• STEP 4 : Start and enable NFS server
    systemctl restart nfs-kernel-server
    systemctl enable nfs-kernel-server
    (if systemctl does not work:  service nfs-kernel-server restart)

• STEP 5 : Mount the share as a client (using localhost)
    mkdir -p /mnt/nfs-client
    mount -t nfs localhost:/srv/nfs/share /mnt/nfs-client
    df -h /mnt/nfs-client





• STEP 6 : Upload files through NFS
    echo "File 1 uploaded via NFS" > /mnt/nfs-client/file1.txt
    echo "File 2 uploaded via NFS" > /mnt/nfs-client/file2.txt

    cp /etc/hostname /mnt/nfs-client/hostname.txt
    mkdir /mnt/nfs-client/docs
    echo "inside docs folder" > /mnt/nfs-client/docs/note.txt
    ls -l /mnt/nfs-client



• STEP 7 : Web browser viewer (nginx autoindex)
    cat > /etc/nginx/conf.d/nfs-viewer.conf << 'EOF'
    server {
        listen 8080;
        server_name _;
        root /mnt/nfs-client;
        autoindex on;
        autoindex_exact_size off;
        autoindex_localtime on;
    }
    EOF


    sed -i 's/^    //' /etc/nginx/conf.d/nfs-viewer.conf
    nginx -t
    systemctl restart nginx



• STEP 8 : Open in the browser
    Killercoda top menu -> "Traffic / Ports" -> Port 8080 -> Access
    You will see the file list (welcome.txt, file1.txt, docs/ ...).
    Or test from terminal:  curl http://localhost:8080/




================================================================
VERIFY CONFIGURATION
================================================================

• CHECK 1 : NFS service running
    systemctl status nfs-kernel-server --no-pager
    -> active (exited) or active (running) is OK

• CHECK 2 : NFS port 2049 listening
    ss -tlnp | grep 2049

• CHECK 3 : RPC services registered
    rpcinfo -p | grep -E "nfs|mountd|portmapper"

• CHECK 4 : Export is active
    exportfs -v
    showmount -e localhost
    -> /srv/nfs/share must be listed

• CHECK 5 : Share is mounted
    mount | grep nfs
    findmnt /mnt/nfs-client

• CHECK 6 : Upload check (client -> server)
    ls -l /srv/nfs/share
    -> file1.txt, file2.txt, hostname.txt, docs/ must be here
       (files created on the mount appear in the real server folder)

• CHECK 7 : Two-way check (server -> client)
    echo "created on server side" > /srv/nfs/share/from_server.txt
    cat /mnt/nfs-client/from_server.txt

• CHECK 8 : Download / read check
    cat /mnt/nfs-client/welcome.txt
    -> Hello from NFS server

• CHECK 9 : Web viewer shows the files
    curl -s http://localhost:8080/ | grep -E "file1|file2|welcome|docs"
    curl -s http://localhost:8080/welcome.txt

• CHECK 10 : NFS statistics
    nfsstat -s | head -20
    nfsstat -m




================================================================
ONE-SHOT VERIFY SCRIPT
================================================================
    cat > /root/verify_nfs.sh << 'EOF'
    #!/bin/bash
    ok(){ echo "[PASS] $1"; }
    no(){ echo "[FAIL] $1"; }
    systemctl is-active --quiet nfs-kernel-server && ok "nfs-server active" || no "nfs-server active"
    ss -tln | grep -q ':2049 ' && ok "port 2049 listening" || no "port 2049 listening"
    showmount -e localhost 2>/dev/null | grep -q /srv/nfs/share && ok "export visible" || no "export visible"
    mountpoint -q /mnt/nfs-client && ok "client mounted" || no "client mounted"
    echo "verify $(date +%s)" > /mnt/nfs-client/verify.txt && ok "write via NFS" || no "write via NFS"
    diff /mnt/nfs-client/verify.txt /srv/nfs/share/verify.txt >/dev/null && ok "upload reached server folder" || no "upload reached server folder"
    curl -s http://localhost:8080/ | grep -q verify.txt && ok "web viewer lists uploaded file" || no "web viewer lists uploaded file"
    curl -s http://localhost:8080/welcome.txt | grep -q "Hello from NFS" && ok "web viewer reads file" || no "web viewer reads file"
    EOF
    sed -i 's/^    //' /root/verify_nfs.sh
    chmod +x /root/verify_nfs.sh
    /root/verify_nfs.sh


================================================================
EXTRA : AUTO-MOUNT AT BOOT (optional)
================================================================
    echo "localhost:/srv/nfs/share /mnt/nfs-client nfs defaults 0 0" >> /etc/fstab
    mount -a


================================================================
TROUBLESHOOTING
================================================================
• "mount.nfs: access denied"        -> check /etc/exports, then:  exportfs -ra
• "Connection refused"              -> systemctl restart nfs-kernel-server rpcbind
• "Permission denied" when writing  -> chmod 777 /srv/nfs/share  (and keep no_root_squash)
• "Stale file handle"               -> umount -f /mnt/nfs-client ; mount again
• nginx 403 Forbidden               -> chmod -R o+rX /mnt/nfs-client
• nginx port in use                 -> change 8080 in nfs-viewer.conf to 8081
• nfs-kernel-server fails to start  -> journalctl -u nfs-kernel-server -n 30
    (If the kernel NFS server is not supported in your environment, use
     the user-space alternative:  apt install nfs-ganesha)


================================================================
CLEANUP
================================================================
    umount /mnt/nfs-client
    sed -i '\|/srv/nfs/share|d' /etc/exports
    sed -i '\|/mnt/nfs-client|d' /etc/fstab
    exportfs -ra
    rm -f /etc/nginx/conf.d/nfs-viewer.conf
    systemctl restart nginx
    systemctl stop nfs-kernel-server
    rm -rf /srv/nfs

SAMBA SERVER + WEB FILE VIEWER - KILLERCODA UBUNTU - SIMPLE STEPS
==================================================================
Server share : /srv/samba/share      (shared as "labshare")
Client mount : /mnt/smb-client       (mounted from localhost over CIFS/SMB)
Web viewer   : nginx on port 8081    (shows the files from the Samba mount)
Samba user   : smbuser / smb123




Flow:  upload file -> Samba mount -> stored in server share -> visible in browser





• STEP 1 : Become root and install packages
    sudo -i
    apt update
    apt install -y samba smbclient cifs-utils nginx
    apt install samba-common-bin


• STEP 2 : Create the share folder and a test file
    mkdir -p /srv/samba/share
    echo "Hello from Samba server" > /srv/samba/share/welcome.txt



• STEP 3 : Create the Samba user
    useradd -M -s /usr/sbin/nologin smbuser
    printf 'smb123\nsmb123\n' | smbpasswd -a -s smbuser
    smbpasswd -e smbuser
    chown -R smbuser:smbuser /srv/samba/share
    chmod -R 2775 /srv/samba/share



• STEP 4 : Configure the share
    cp /etc/samba/smb.conf /etc/samba/smb.conf.bak
    cat >> /etc/samba/smb.conf << 'EOF'

    [labshare]
        comment = Lab Samba Share
        path = /srv/samba/share
        browseable = yes
        read only = no
        valid users = smbuser
        create mask = 0664
        directory mask = 0775
    EOF


    sed -i 's/^    //' /etc/samba/smb.conf
    testparm -s





  Meaning of options:
    path            folder that is shared
    browseable      share is visible in the share list
    read only = no  read and write allowed
    valid users     only smbuser may connect
    create mask     permission of new files
    directory mask  permission of new folders

  (If sed removes the indent of the whole file, no problem: smb.conf works
   with or without indentation.)






• STEP 5 : Start and enable Samba
    systemctl restart smbd nmbd
    systemctl enable smbd nmbd
    (if systemctl does not work:  service smbd restart)

• STEP 6 : Mount the share as a client (using localhost)
    mkdir -p /mnt/smb-client
    mount -t cifs //localhost/labshare /mnt/smb-client -o username=smbuser,password=smb123,vers=3.0
    df -h /mnt/smb-client



• STEP 7 : Upload files through Samba
    echo "File 1 uploaded via Samba" > /mnt/smb-client/file1.txt
    echo "File 2 uploaded via Samba" > /mnt/smb-client/file2.txt
    cp /etc/hostname /mnt/smb-client/hostname.txt
    mkdir /mnt/smb-client/docs
    echo "inside docs folder" > /mnt/smb-client/docs/note.txt
    ls -l /mnt/smb-client


  (Upload without mounting, using smbclient:)
    echo "uploaded by smbclient" > /tmp/up.txt
    smbclient //localhost/labshare -U smbuser%smb123 -c "put /tmp/up.txt up.txt"


• STEP 8 : Web browser viewer (nginx autoindex)

==================================================================
ONE-SHOT VERIFY SCRIPT
==================================================================
    cat > /root/verify_samba.sh << 'EOF'
    #!/bin/bash
    ok(){ echo "[PASS] $1"; }
    no(){ echo "[FAIL] $1"; }
    systemctl is-active --quiet smbd && ok "smbd active" || no "smbd active"
    ss -tln | grep -q ':445 ' && ok "port 445 listening" || no "port 445 listening"
    testparm -s 2>/dev/null | grep -q "\[labshare\]" && ok "share defined" || no "share defined"
    pdbedit -L | grep -q smbuser && ok "samba user exists" || no "samba user exists"
    smbclient -L localhost -U smbuser%smb123 2>/dev/null | grep -q labshare && ok "share visible" || no "share visible"
    mountpoint -q /mnt/smb-client && ok "client mounted" || no "client mounted"
    echo "verify $(date +%s)" > /mnt/smb-client/verify.txt && ok "write via Samba" || no "write via Samba"
    diff /mnt/smb-client/verify.txt /srv/samba/share/verify.txt >/dev/null && ok "upload reached server folder" || no "upload reached server folder"
    smbclient //localhost/labshare -U smbuser%wrong -c ls >/dev/null 2>&1 && no "wrong password blocked" || ok "wrong password blocked"
    curl -s http://localhost:8081/ | grep -q verify.txt && ok "web viewer lists uploaded file" || no "web viewer lists uploaded file"
    curl -s http://localhost:8081/welcome.txt | grep -q "Hello from Samba" && ok "web viewer reads file" || no "web viewer reads file"
    EOF
    sed -i 's/^    //' /root/verify_samba.sh
    chmod +x /root/verify_samba.sh
    /root/verify_samba.sh

==================================================================
TROUBLESHOOTING
==================================================================
• "mount error(13): Permission denied" -> wrong user/password; reset:
                                          printf 'smb123\nsmb123\n' | smbpasswd -a -s smbuser
• "mount error(95)/(2) no such device"  -> cifs module missing; use smbclient
                                          commands instead (they need no mount)
• "NT_STATUS_ACCESS_DENIED" on write    -> chown -R smbuser:smbuser /srv/samba/share
• "NT_STATUS_BAD_NETWORK_NAME"          -> share name wrong; check testparm -s
• testparm shows errors                 -> fix /etc/samba/smb.conf, restart smbd
• nginx 403 Forbidden                   -> chmod -R o+rX /mnt/smb-client
• nginx port in use                     -> change 8081 in smb-viewer.conf to 8082
• Samba logs                            -> tail -n 30 /var/log/samba/log.smbd

    cp /etc/samba/smb.conf.bak /etc/samba/smb.conf
    smbpasswd -x smbuser
    userdel smbuser
    rm -f /etc/nginx/conf.d/smb-viewer.conf
    systemctl restart nginx
    systemctl stop smbd nmbd
    rm -rf /srv/samba
