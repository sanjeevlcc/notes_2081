------------------------------------------------------------
  STEP 4: Copy and run Cell 3 — Run the Simulation

  . . . . . . . . . . . . . . . . . . . . . . . . . . . .
  # CELL 3: SIMULATE WATERFALL EXECUTION
  # Simulate each phase with random variation (real projects vary)

  results = []
  total_planned = 0
  total_actual  = 0
  cumulative_weeks = 0
  total_defects_injected = 0
  total_defects_found    = 0

  for phase in phases:
      planned = phase["planned_weeks"]
      tasks   = phase["tasks_total"]

      # Actual time varies +/- 50% of planned (real-world variation)
      variation    = np.random.uniform(0.8, 1.6)
      actual_weeks = round(planned * variation, 1)

      # Defects: injected based on tasks, some detected, some slip
      defects_injected = int(tasks * phase["defect_rate"])
      defects_detected = int(defects_injected * phase["detection_rate"])
      defects_slipped  = defects_injected - defects_detected

      # Rework: each detected defect costs extra time
      rework_weeks = defects_detected * 0.05  # 0.05 weeks per defect
      actual_weeks_with_rework = round(actual_weeks + rework_weeks, 1)

      cumulative_weeks       += actual_weeks_with_rework
      total_planned          += planned
      total_actual           += actual_weeks_with_rework
      total_defects_injected += defects_injected
      total_defects_found    += defects_detected

      results.append({
          "Phase":             phase["name"],
          "Planned (wks)":     planned,
          "Actual (wks)":      actual_weeks_with_rework,
          "Variance":          round(actual_weeks_with_rework - planned, 1),
          "Tasks":             tasks,
          "Defects Injected":  defects_injected,
          "Defects Found":     defects_detected,
          "Defects Slipped":   defects_slipped,
          "Cumulative Wks":    round(cumulative_weeks, 1)
      })

  df_waterfall = pd.DataFrame(results)
  print("WATERFALL SIMULATION RESULTS")
  print("=" * 75)
  print(df_waterfall.to_string(index=False))
  print("=" * 75)
  print(f"Total Planned  : {total_planned:.0f} weeks")
  print(f"Total Actual   : {total_actual:.1f} weeks")
  print(f"Overrun        : {total_actual - total_planned:.1f} weeks "
        f"({(total_actual/total_planned - 1)*100:.1f}%)")
  print(f"Defects Injected: {total_defects_injected}")
  print(f"Defects Found   : {total_defects_found}")
  print(f"Defects Slipped : {total_defects_injected - total_defects_found}")

------------------------------------------------------------

  . . . . . . . . . . . . . . . . . . . . . . . . . . . .
  # CELL 3: SIMULATE INCREMENTAL EXECUTION

  inc_results = []
  cumulative  = 0
  base_defect_rate = 0.20  # Starts high, drops with experience

  for inc in increments:
      planned = inc["planned_weeks"]
      tasks   = inc["tasks"]
      lf      = inc["learning_factor"]

      # Actual time reduces with learning factor
      # More experienced team = faster delivery
      variation    = np.random.uniform(0.85, 1.30)
      actual_weeks = round((planned / lf) * variation, 1)

      # Defect rate improves with each increment (team learns)
      defect_rate      = base_defect_rate / lf
      defects_injected = int(tasks * defect_rate)

      # Customer feedback: 20% chance of scope change per increment
      scope_change = np.random.random() < 0.20
      extra_tasks  = int(tasks * 0.15) if scope_change else 0
      if scope_change:
          actual_weeks += 1.0  # Scope change costs 1 extra week

      # Customer satisfaction: improves with each increment
      satisfaction = min(5.0, 2.5 + (inc["id"] * 0.5)
                         + np.random.uniform(-0.3, 0.3))

      cumulative += actual_weeks

      inc_results.append({
          "Increment":     inc["id"],
          "Name":          inc["name"],
          "Planned (wks)": planned,
          "Actual (wks)":  actual_weeks,
          "Variance":      round(actual_weeks - planned, 1),
          "Tasks Done":    tasks + extra_tasks,
          "Defect Rate":   f"{defect_rate*100:.1f}%",
          "Defects":       defects_injected,
          "Scope Change":  "YES" if scope_change else "NO",
          "Satisfaction":  round(satisfaction, 1),
          "Cumulative":    round(cumulative, 1)
      })

  df_inc = pd.DataFrame(inc_results)
  print("INCREMENTAL MODEL SIMULATION RESULTS")
  print("=" * 80)
  print(df_inc.to_string(index=False))
  print("=" * 80)
  total_weeks = df_inc["Actual (wks)"].sum()
  avg_sat     = df_inc["Satisfaction"].mean()
  print(f"Total Duration      : {total_weeks:.1f} weeks")
  print(f"Avg Satisfaction    : {avg_sat:.1f} / 5.0")
  print(f"Defect Rate Trend   : Improving across increments (learning)")
  . . . . . . . . . . . . . . . . . . . . . . . . . . . .

------------------------------------------------------------
  STEP 3: Create the waterfall simulation script
------------------------------------------------------------

  . Type the following to create the script file:

  . . . . . . . . . . . . . . . . . . . . . . . . . . . .
  cat > ~/waterfall_sim/waterfall.sh << 'SCRIPT'
  #!/bin/bash

  # ================================================
  # WATERFALL MODEL SIMULATION SCRIPT
  # Simulates 6 phases of software development
  # ================================================

  PROJECT="LibrarySystem"
  LOG="./logs/project_log.txt"
  DATE=$(date "+%Y-%m-%d %H:%M:%S")

  # Initialize log file
  echo "================================================" > $LOG
  echo " WATERFALL SIMULATION: $PROJECT"               >> $LOG
  echo " Started: $DATE"                               >> $LOG
  echo "================================================" >> $LOG

  # ---- Helper function: simulate phase ----
  run_phase() {
      local PHASE=$1
      local PLANNED=$2   # planned weeks
      local TASKS=$3     # total tasks

      echo ""
      echo "========================================"
      echo " PHASE: $PHASE"
      echo " Planned Duration: $PLANNED weeks"
      echo " Total Tasks: $TASKS"
      echo "========================================"

      # Simulate work (random variation +/- 30%)
      VARIATION=$(echo "scale=2; $PLANNED * (0.8 + $RANDOM % 50 / 100.0)" | bc)
      ACTUAL=$(echo "scale=1; $VARIATION" | bc)

      # Simulate tasks completed
      COMPLETED=$(echo "$TASKS - ($TASKS * $RANDOM / 100000)" | bc)

      # Simulate defects found
      DEFECTS=$(echo "$TASKS / 10 + $RANDOM % 5" | bc)

      # Write to log
      echo "---" >> $LOG
      echo "Phase     : $PHASE"           >> $LOG
      echo "Planned   : $PLANNED weeks"   >> $LOG
      echo "Actual    : $ACTUAL weeks"    >> $LOG
      echo "Tasks     : $COMPLETED/$TASKS" >> $LOG
      echo "Defects   : $DEFECTS found"   >> $LOG

      # Create phase artifact file
      echo "$PHASE complete. Tasks:$COMPLETED. Defects:$DEFECTS." \
           > ./artifacts/${PHASE}.txt

      # Wait briefly (simulates work happening)
      sleep 1

      echo " Status: COMPLETE"
      echo " Artifact saved: ./artifacts/${PHASE}.txt"
  }

  # ---- Run all 6 Waterfall phases in sequence ----
  echo "Starting Waterfall Simulation..."
  echo ""

  run_phase "Requirements"    4  50

  STEP 6: Now simulate Incremental model
------------------------------------------------------------

  . . . . . . . . . . . . . . . . . . . . . . . . . . . .
  cat > ~/waterfall_sim/incremental.sh << 'SCRIPT'
  #!/bin/bash

  # ================================================
  # INCREMENTAL MODEL SIMULATION SCRIPT
  # Simulates 4 increments, each delivering software
  # ================================================

  LOG="./logs/incremental_log.txt"
  echo "INCREMENTAL MODEL SIMULATION" > $LOG
  echo "============================" >> $LOG

  run_increment() {
      local NUM=$1
      local NAME=$2
      local WEEKS=$3
      local FEATURES=$4

      echo ""
      echo "========================================"
      echo " INCREMENT $NUM: $NAME"
      echo " Duration: $WEEKS weeks"
      echo " Features: $FEATURES"
      echo "========================================"

      # Learning factor improves with each increment
      LEARN=$(echo "scale=2; 1 + ($NUM - 1) * 0.15" | bc)
      ACTUAL=$(echo "scale=1; $WEEKS / $LEARN" | bc)

      echo " Learning Factor : ${LEARN}x"
      echo " Planned Duration: $WEEKS weeks"
      echo " Actual Duration : $ACTUAL weeks (faster due to learning)"

      # Satisfaction increases with each increment
      SAT=$(echo "scale=1; 2.5 + $NUM * 0.5" | bc)
      echo " Customer Satisfaction: $SAT / 5.0"

      echo "---"                          >> $LOG
      echo "Increment $NUM: $NAME"        >> $LOG
      echo "  Planned : $WEEKS weeks"     >> $LOG
      echo "  Actual  : $ACTUAL weeks"    >> $LOG
      echo "  Satisfaction: $SAT/5.0"     >> $LOG
      echo "  Features: $FEATURES"        >> $LOG

      # Create deliverable
      echo "INCREMENT $NUM DELIVERED: $NAME ($FEATURES)" \
           > ./artifacts/increment_${NUM}.txt

      sleep 1
      echo " Delivered to customer. Feedback collected."
  }

  echo "Starting Incremental Simulation..."
  run_increment 1 "Core Features"     8  "Login,Search,Profiles"
  run_increment 2 "Main Features"     8  "Reports,Issue,Return"
  run_increment 3 "Advanced Features" 8  "Analytics,Notifications"
  run_increment 4 "Final Polish"      8  "Admin,Export,Performance"

  echo ""
  echo "INCREMENTAL SIMULATION COMPLETE"
  echo "Deliverables:"
  ls -lh ./artifacts/increment_*.txt
  echo ""
  echo "Full Log:"
  cat $LOG
  SCRIPT

  chmod +x ~/waterfall_sim/incremental.sh
  bash ~/waterfall_sim/incremental.sh
  . . . . . . . . . . . . . . . . . . . . . . . . . . . .









============================================================
  LAB 5: DOCKER — EACH PHASE AS A CONTAINER
============================================================

  TOOL     : Docker on KillerCoda or local Ubuntu
  TIME     : 30-40 minutes
  CONCEPTS : Each phase = isolated Docker container,
             containers pass artifacts between phases,
             demonstrates real CI/CD pipeline concept

------------------------------------------------------------
  WHY DOCKER FOR THIS LAB?
------------------------------------------------------------

  . In real software projects, each phase often runs in
    an ISOLATED ENVIRONMENT:
    . Requirements: collaboration tools
    . Development: coding environments
    . Testing: test runner environments
    . Deployment: production containers
  . Docker containers are PERFECT for modeling this.
  . Each container = one phase = one isolated environment.
  . Containers share a VOLUME (like a shared file system)
    to pass artifacts from one phase to the next.

------------------------------------------------------------
  STEP 1: Verify Docker is available
------------------------------------------------------------

  . . . . . . . . . . . . . . . . . . . . . . . . . . . .
  docker --version
  docker ps

  . On KillerCoda Docker playground: Docker is pre-installed.
  . On Ubuntu: sudo apt install docker.io -y

  STEP 2: Simulate Waterfall with Git branches

      git merge "increment/$NAME" \
          -m "RELEASE $i: $NAME delivered to customer"
      echo "Increment $i complete and delivered."
  done
