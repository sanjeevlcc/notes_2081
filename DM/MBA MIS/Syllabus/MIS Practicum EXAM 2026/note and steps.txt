df_clean = df.copy()

# Remove duplicates
df_clean = df_clean.drop_duplicates()
print("After removing duplicates:", df_clean.shape)

# Fix impossible values
df_clean.loc[df_clean["monthly_data_gb"] > 9000, "monthly_data_gb"] = np.nan
df_clean["monthly_data_gb"].fillna(df_clean["monthly_data_gb"].median(), inplace=True)

# Fill missing decision_confidence
df_clean["decision_confidence"].fillna(df_clean["decision_confidence"].median(), inplace=True)

# Normalize text
df_clean["industry"] = df_clean["industry"].str.strip().str.title()
df_clean["data_trust"] = df_clean["data_trust"].str.strip().str.title()
df_clean["cynefin_domain"] = df_clean["cynefin_domain"].str.strip().str.title()

===========================================================================
  STEP 10 — VISUALIZATION 3: DATA QUALITY vs DECISION CONFIDENCE (SCATTER)

colors_map = {"Low":"#e34948","Medium":"#eda100","High":"#1baf7a"}
colors_scatter = df_clean["data_trust"].map(colors_map)

fig, ax = plt.subplots(figsize=(9, 6))
ax.scatter(df_clean["data_quality_score"],
           df_clean["decision_confidence"],
           c=colors_scatter, alpha=0.35, s=15)

for label, color in colors_map.items():
    ax.scatter([], [], color=color, label=f"Trust: {label}", s=40)

z = np.polyfit(df_clean["data_quality_score"].dropna(),
               df_clean["decision_confidence"].dropna(), 1)
p = np.poly1d(z)
x_line = np.linspace(1, 10, 100)
ax.plot(x_line, p(x_line), "#2a78d6", linewidth=2,
        linestyle="--", label="Trend")

ax.set_title("Data Quality vs Decision Confidence\n"
             "(Article Ref: Trust Theory — Moore, 2017)",
             fontsize=13, fontweight="bold")

corr_cols = [
    "data_quality_score","viz_clarity_score",
    "decision_confidence","decision_speed_score",
    "collaboration_score","d3m_adoption_score",
    "big_data_readiness","pct_decisions_datadriven"
]

corr = df_clean[corr_cols].corr().round(2)

fig, ax = plt.subplots(figsize=(9, 7))
cax = ax.imshow(corr, cmap="RdBu_r", vmin=-1, vmax=1)
plt.colorbar(cax, ax=ax, fraction=0.046, label="Correlation r")

labels = ["Data Quality","Viz Clarity","Confidence","Speed",
          "Collaboration","D3M Score","Big Data Ready","% D3M Decisions"]
ax.set_xticks(range(len(corr_cols)))
ax.set_yticks(range(len(corr_cols)))
ax.set_xticklabels(labels, rotation=45, ha="right", fontsize=9)
ax.set_yticklabels(labels, fontsize=9)

for i in range(len(corr_cols)):
    for j in range(len(corr_cols)):
        val = corr.iloc[i, j]
        color = "white" if abs(val) > 0.5 else "black"
        ax.text(j, i, f"{val}", ha="center",
                va="center", fontsize=7.5, color=color)

ax.set_title("Correlation Heatmap — MIS522 Survey\n"
             "(Article Ref: CFT + D3M — Moore, 2017)",
             fontsize=13, fontweight="bold", pad=15)
plt.tight_layout()
